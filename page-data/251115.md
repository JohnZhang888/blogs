## T1

如果这道题的 $n \le 3000$，就可以把所有的节点都连上边然后跑最短路，这与 [CF1245D](https://codeforces.com/contest/1245/problem/D) 的解法比较类似。但这里 $n$ 非常大，我们需要选择性建边。

仔细观察这个有些古怪的距离表达式。可以发现，假如我们有三个点 $A(1, y_1), B(2, y_2), C(3, y_3)$ （暂时先不考虑纵坐标），那么从 $A$ 走到 $B$，再从 $B$ 走到 $C$，一定不会比从 $A$ 直接走到 $C$ 更劣。因此我们可以先将所有的点按照其横坐标排序，然后在相邻的点之间建边，边权为两个点的横坐标之差。纵坐标可以按同样的方式处理。然后再加上本来就有的隧道，跑 Dijkstra 即可。

[跳过代码](#t2)

```cpp
#include <bits/stdc++.h>
#define int long long
const int N = 2e5+10, M = 2e5+10;
struct Node {
  int x, y, id;
} nodes[N];
struct Edge {
  int to, len;
};
std::vector<Edge> g[N];
int n, m;
struct DijkNode {
  int u, dis;
  bool operator > (const DijkNode &a) const {
    return dis > a.dis;
  }
};
int dis[N]; bool vis[N];
inline int read() {
  int x = 0, f = 1;
  char ch = getchar();
  while (ch < '0' || ch > '9') {
    if (ch == '-') f = -1;
    ch = getchar();
  }
  while (ch >= '0' && ch <= '9') {
    x = x * 10 + ch - '0', ch = getchar();
  }
  return x * f;
}
void write(int x) {
  if (x < 0) {
    putchar('-');
    x = -x;
  }
  if (x > 9) {
    write(x / 10);
  }
  putchar(x % 10 + '0');
  return;
}

signed main() {
  freopen("teleport.in", "r", stdin);
  freopen("teleport.out", "w", stdout);
  std::ios::sync_with_stdio(false); std::cin.tie(0);
  n = read(), m = read();
  for (int i = 1; i <= n; i++) {
    nodes[i].x = read(), nodes[i].y = read();
    nodes[i].id = i;
  }
  for (int i = 1; i <= m; i++) {
    int u, v, w;
    u = read(), v = read(), w = read();
    g[u].push_back({v, w});
    g[v].push_back({u, w});
  }
  std::sort(nodes+1, nodes+1+n, [](Node a, Node b) {
    return a.x < b.x;
  });
  for (int i = 1; i <= n - 1; i++) {
    g[nodes[i].id].push_back({nodes[i+1].id, nodes[i+1].x - nodes[i].x});
    g[nodes[i+1].id].push_back({nodes[i].id, nodes[i+1].x - nodes[i].x});
  }
  std::sort(nodes+1, nodes+1+n, [](Node a, Node b) {
    return a.y < b.y;
  });
  for (int i = 1; i <= n - 1; i++) {
    g[nodes[i].id].push_back({nodes[i+1].id, nodes[i+1].y - nodes[i].y});
    g[nodes[i+1].id].push_back({nodes[i].id, nodes[i+1].y - nodes[i].y});
  }
  for (int i = 1; i <= n; i++) {
    dis[i] = 1e18;
  }
  dis[1] = 0;
  std::priority_queue<DijkNode, std::vector<DijkNode>, std::greater<DijkNode> > q;
  q.push({1, 0});
  while (!q.empty()) {
    int u = q.top().u;
    q.pop();
    if (vis[u]) continue;
    vis[u] = true;
    for (const auto &v : g[u]) {
      int to = v.to, len = v.len;
      if (dis[u] + len < dis[to]) {
        dis[to] = dis[u] + len;
        q.push({to, dis[to]});
      }
    }
  }
  for (int i = 2; i <= n; i++) {
    write(dis[i]); putchar(' ');
  }
  return 0;
}
```

## T2

这题官方题解是 DP，但我觉得很不优雅。于是刚才问了一下 DeepSeek，找到了一种排列组合的解法。

这道题的关键在于处理 $k$ 的倍数（关键数字）之间的相邻关系。如果两个关键数字相邻且它们的最大公约数为 $k$，则该排列无效。明显地，**如果两个数不都是 $k$ 的倍数，则两者的最大公约数一定不是 $k$**，这就是本题的关键。

关键数字的数量 $m = \lfloor \dfrac{n}{k} \rfloor$。考虑到这些数字的数量不会超过 $10$，我们可以通过 `std::next_permutation` 来枚举所有排列，并针对每一个排列，计算其中相邻关键数字对的最大公约数为 $k$ 的数量 $c(\pi)$。对于每个排列 $\pi$，计算从 $n$ 个位置中选择 $m$ 个位置防止关键数字，且禁止相邻对的数量为 $c(\pi)$ 的方案数，即 $C(n-c(\pi), m)$。最后，将所有排列的 $C(n-c(\pi), m)$ 相加，并乘以非关键数字的排列数 $(n - m)!$。

[跳过代码](#t3)

```cpp
#include <bits/stdc++.h>
typedef long long ll;
const int N = 3e3+10, MOD = 998244353;
int n, k;
ll fact[N], infact[N], a[15], sum;
ll qpow(ll a, ll b) {
  ll res = 1;
  while (b) {
    if (b & 1) res = res * a % MOD;
    a = a * a % MOD;
    b >>= 1;
  }
  return res;
}
void init() {
  fact[0] = infact[0] = 1;
  for (int i = 1; i <= n; i++) {
    fact[i] = fact[i-1] * i % MOD;
    infact[i] = infact[i-1] * qpow(i, MOD - 2) % MOD;
  }
}
ll getC(int n, int m) {
  return (fact[n] * infact[n - m] % MOD * infact[m] % MOD) % MOD;
}

int main() {
  freopen("permutation.in", "r", stdin);
  freopen("permutation.out", "w", stdout);
  std::ios::sync_with_stdio(false); std::cin.tie(0);  
  std::cin >> n >> k;
  init();
  int m = n / k;
  if (m == 0) {
    std::cout << fact[n] << '\n';
    return 0;
  }
  for (int i = 1; i <= m; i++) {
    a[i] = i;
  }
  ll sum = 0;
  do {
    int c = 0;
    for (int i = 1; i <= m - 1; i++) {
      if (std::__gcd(a[i], a[i+1]) == 1) c++; 
    }
    sum = (sum + getC(n-c, m)) % MOD;
  } while (std::next_permutation(a+1, a+1+m));
  ll ans = fact[n-m] * sum % MOD;
  std::cout << ans << '\n';
  return 0;
}
```

## T3

这道题场上想到了线段树做法，但是跟上次一样，因为节点设计不合理而遗憾离场。题解给出了一个很天才的想法：维护活着的（包括濒死的）人的最小值、最小值个数和死亡人数；对于护盾则直接暴力到叶子进行维护。当区间最小值为 $0$ 时，区间最小值个数就是濒死人数。另外，当区间内有人死亡时，可以直接将最小值设置为 $+\infty$，避免对统计最小值造成影响。

```cpp
#include <bits/stdc++.h>
#define int long long
typedef long long ll;
const int N = 2e5+10;
int n, q, a[N];
struct Segtree {
  struct Node {
    ll minVal, minCnt, death, hpTag, shield; // 练 OI 学英文之 shield n.护盾
  } tr[4 * N];
  void pushUp(int p) {
    tr[p].death = tr[p*2].death + tr[p*2+1].death;
    tr[p].shield = tr[p*2].shield + tr[p*2+1].shield;
    tr[p].minVal = std::min(tr[p*2].minVal, tr[p*2+1].minVal);
    if (tr[p*2].minVal == tr[p*2+1].minVal) {
      tr[p].minCnt = tr[p*2+1].minCnt + tr[p*2].minCnt;
    } else if (tr[p*2].minVal < tr[p*2+1].minVal) {
      tr[p].minCnt = tr[p*2].minCnt;
    } else {
      tr[p].minCnt = tr[p*2+1].minCnt;
    }
  }
  void build(int s = 1, int t = n, int p = 1) {
    if (s == t) {
      tr[p].minVal = a[s];
      tr[p].minCnt = 1;
      return;
    }
    int mid = (s + t) >> 1;
    build(s, mid, p*2);
    build(mid+1, t, p*2+1);
    pushUp(p);
  }
  void pushDown(int s, int t, int p) {
    if (s == t) return;
    if (tr[p].hpTag == 0) return;
    tr[p*2].minVal += tr[p].hpTag;
    tr[p*2+1].minVal += tr[p].hpTag;
    tr[p*2].hpTag += tr[p].hpTag;
    tr[p*2+1].hpTag += tr[p].hpTag;
    tr[p].hpTag = 0;
  }
  void hurt(int l, int r, int c, int s = 1, int t = n, int p = 1) {
    if (s == t) {
      if (tr[p].shield) {
        tr[p].shield--;
        return;
      }
      if (tr[p].minVal < c) {
        tr[p].minVal = 1e18;
        tr[p].death = 1;
        tr[p].shield = 0;
      } else {
        tr[p].minVal -= c;
      }
      return;
    }
    if (l <= s && t <= r && !tr[p].shield && tr[p].minVal >= c) {
      tr[p].minVal -= c, tr[p].hpTag -= c;
      return;
    }    
    pushDown(s, t, p);
    int mid = (s + t) >> 1;
    if (l <= mid) {
      hurt(l, r, c, s, mid, p*2);
    }
    if (r > mid) {
      hurt(l, r, c, mid+1, t, p*2+1);
    }
    pushUp(p);
  }
  void treat(int l, int r, int c, int s = 1, int t = n, int p = 1) {
    if (l <= s && t <= r) {
      tr[p].minVal += c;
      tr[p].hpTag += c;
      return;
    }
    pushDown(s, t, p);
    int mid = (s + t) >> 1;
    if (l <= mid) {
      treat(l, r, c, s, mid, p*2);
    }
    if (r > mid) {
      treat(l, r, c, mid+1, t, p*2+1);
    }
    pushUp(p);
  }
  void addShield(int l, int c = 1, int s = 1, int t = n, int p = 1) {
    if (s == t) {
      if (!tr[p].death) tr[p].shield++;
      return;
    }
    pushDown(s, t, p);
    int mid = (s + t) >> 1;
    if (l <= mid) {
      addShield(l, c, s, mid, p*2);
    } else {
      addShield(l, c, mid+1, t, p*2+1);
    }
    pushUp(p);
  }
  int queryDeath(int l, int r, int s = 1, int t = n, int p = 1) {
    if (l <= s && t <= r) {
      return tr[p].death;
    }
    pushDown(s, t, p);
    int mid = (s + t) >> 1, res = 0;
    if (l <= mid) {
      res += queryDeath(l, r, s, mid, p*2);
    }
    if (r > mid) {
      res += queryDeath(l, r, mid+1, t, p*2+1);
    }
    return res;
  }
  int queryDying(int l, int r, int s = 1, int t = n, int p = 1) {
    if (l <= s && t <= r) {
      if (tr[p].minVal == 0) {
        return tr[p].minCnt;
      } else return 0;
    }
    pushDown(s, t, p);
    int mid = (s + t) >> 1, res = 0;
    if (l <= mid) {
      res += queryDying(l, r, s, mid, p*2);
    }
    if (r > mid) {
      res += queryDying(l, r, mid+1, t, p*2+1);
    }
    return res;
  }
} tree;

signed main() {
  freopen("simulator.in", "r", stdin);
  freopen("simulator.out", "w", stdout);
  std::ios::sync_with_stdio(false); std::cin.tie(0);
  std::cin >> n;
  for (int i = 1; i <= n; i++) {
    std::cin >> a[i];
  }
  tree.build();
  std::cin >> q;
  while (q--) {
    int op;
    std::cin >> op;
    if (op == 1) {
      int l, r, x;
      std::cin >> l >> r >> x;
      tree.hurt(l, r, x);
    } else if (op == 2) {
      int l, r, x;
      std::cin >> l >> r >> x;
      tree.treat(l, r, x);
    } else if (op == 3) {
      int h;
      std::cin >> h;
      tree.addShield(h);
    } else if (op == 4) {
      int l, r;
      std::cin >> l >> r;
      std::cout << tree.queryDeath(l, r) << '\n';
    } else if (op == 5) {
      int l, r;
      std::cin >> l >> r;
      std::cout << tree.queryDying(l, r) << '\n';
    }
  }
  return 0;
}
```
