## T1

第一眼看到这道题就想到容斥，也想到 DP，但是具体怎么做头绪不多，看到 AeeE5x 大神的题解后恍然大悟。

这道题有两个比较重要的限制：

1. **相邻字符不能相同**。这个可以在 DP 时处理。
2. **至少包含一个小写字母、大写字母和数字。** 可以考虑容斥：最后答案的种类数为 `无限制 - 不包含大写字母 - 不包含小写字母 - 不包含数字 + 不包含大写字母和数字 + 不包含大写字母和小写字母 + 不包含小写字母和数字`。

然后考虑如何 DP。我们可以设 $\mathrm{dp}(i, c)$ 表示第 $i$ 个字符为 $c$ 时，当前答案有多少种。对 $\mathrm{dp}(1, c)$ 进行初始化后，则有状态转移方程 $\mathrm{dp}(i, c) = \sum\limits_{c'=0}^{62}\mathrm{dp}(i-1, c'), c \ne c'$。考虑到拮据的空间限制，可以将第一维滚掉。最后返回 $\sum\limits_{c=0}^{62}\mathrm{dp}(|S|, c)$ 即可表示在某种限制条件下的答案数。

为了简化代码，避免写成 AeeE5x 的屎山，可以将刚才提到的 DP 计算过程封装为一个函数，并以三个 `bool` 变量作为参数，分别代表此时是否允许选择小写字母、大写字母和数字。

另外以下代码又使用了我最喜欢的 C++ 特性之一 `enum`。强烈建议大家学一下，这是真正能减轻大脑负担的语法。

[跳过代码](#t2)

```cpp
#include <bits/stdc++.h>
typedef long long ll;
const int N = 1e5+10, MOD = 998244353;
int dp[2][70];
int n, a[N];

enum Type {
  low, upper, num, ques
};

Type getType(int x) {
  if (0 <= x && x < 26) return low;
  else if (26 <= x && x < 52) return upper;
  else if (52 <= x && x < 62) return num;
  else return ques;
}

int getNum(char ch) {
  if (ch >= 'a' && ch <= 'z') return ch - 'a';
  else if (ch >= 'A' && ch <= 'Z') return ch - 'A' + 26;
  else if (ch >= '0' && ch <= '9') return ch - '0' + 52;
  else return 62;
}

ll calc(bool isLow, bool isUpper, bool isNum) {
  if (getType(a[1]) == low) {
    if (isLow) {
      dp[1][a[1]] = 1;
    }
    if (isUpper) {
      dp[1][a[1] + 26] = 1;
    }
  } else if (isUpper && getType(a[1]) == upper) {
    dp[1][a[1]] = 1;    
  } else if (isNum && getType(a[1]) == num) {
    dp[1][a[1]] = 1;
  } else if (getType(a[1]) == ques) {
    for (int i = 0; i < 62; i++) {
      if ((isLow && getType(i) == low) || (isUpper && getType(i) == upper) || (isNum && getType(i) == num)) {
        dp[1][i] = 1;
      }
    }    
  }  
  ll lastSum = 0;
  for (int i = 0; i < 62; i++) {
    if ((isLow && getType(i) == low) || (isUpper && getType(i) == upper) || (isNum && getType(i) == num)) {
      lastSum += dp[1][i];
    }
  }
  for (int i = 2; i <= n; i++) {
    for (int j = 0; j < 62; j++) dp[i&1][j] = 0;
    if (getType(a[i]) == low) {
      if (isLow) {
        dp[i&1][a[i]] = (lastSum - dp[~i&1][a[i]] + MOD) % MOD;
      }
      if (isUpper) {
        dp[i&1][a[i] + 26] = (lastSum - dp[~i&1][a[i] + 26] + MOD) % MOD;
      }
    } else if ((isUpper && getType(a[i]) == upper) || (isNum && getType(a[i]) == num)) {
      dp[i&1][a[i]] = (lastSum - dp[~i&1][a[i]] + MOD) % MOD;
    } else if (getType(a[i]) == ques) {
      for (int j = 0; j < 62; j++) {
        if ((getType(j) == low && isLow) || (getType(j) == upper && isUpper) || (getType(j) == num && isNum)) {
          dp[i&1][j] = (lastSum - dp[~i&1][j] + MOD) % MOD;
        }
      }
    }
    lastSum = 0;
    for (int j = 0; j < 62; j++) {
      if ((getType(j) == low && isLow) || (getType(j) == upper && isUpper) || (getType(j) == num && isNum)) {
        lastSum = (lastSum + dp[i&1][j] + MOD) % MOD;
      }
    }
  }  
  std::memset(dp, 0, sizeof(dp));
  return lastSum;
}

int main() {
  freopen("password.in", "r", stdin);
  freopen("password.out", "w", stdout);
  std::ios::sync_with_stdio(false); std::cin.tie(0);
  std::cin >> n;
  for (int i = 1; i <= n; i++) {
    char ch;
    std::cin >> ch;
    a[i] = getNum(ch);
  }
  for (int i = 1; i <= n; i++) {
    std::cerr << a[i] << ' ';
  }
  std::cerr << '\n';
  ll ans1 = calc(true, true, true);
  ll ans2 = ((calc(true, true, false) + calc(true, false, true) + calc(false, true, true)) + MOD) % MOD;
  ll ans3 = ((calc(true, false, false) + calc(false, true, false) + calc(false, false, true)) + MOD) % MOD;
  std::cout << ((ans1 - ans2 + MOD) % MOD + ans3 + MOD) % MOD << '\n';
  std::cerr << ans1 % MOD << ' ' << ans2 % MOD << ' ' << ans3 % MOD << '\n';
  return 0;
}
```

## T2

很明显，这是一道分类讨论题，而讨论的方式直接决定了思考复杂度和码量。

我们可以分以下两种情况来讨论：

1. **小 Z 更早到达 $T$ 点**。那么一定有 $p = T, t = \mathrm{dis}(S', T)$。
2. **小 Z 更晚，或者与小 Y 同时到达 $T$ 点**。考虑 $S' \rightarrow T$ 和 $S \rightarrow T$ 重合的一段，小 Z 可以选择刚到这段时转身向 $S$ 走，和小 Y 在途中相遇。而这段重合的路径正是 $S' \rightarrow S$，那么 $t = \lceil\dfrac{\mathrm{dis}(S, S')}{2}\rceil$。至于 $p$，由于当路径长度为奇数时，小 Y 会在距离小 Z 一条边的时候停下来 $1$ 秒，所以实现略微复杂，具体可以看代码。

另外，为了实现本题的一些功能，需要实现快速求 LCA 和第 $k$ 个祖先的函数，以及求两点距离的函数，可以用倍增实现。

```cpp
#include <bits/stdc++.h>
typedef long long ll;
const int N = 2e5+10;
std::vector<int> g[N];
int n, q, f[N][21], dep[N];

void dfs(int u, int pre) {
  for (auto &v : g[u]) {
    if (v == pre) continue;
    dep[v] = dep[u] + 1;
    f[v][0] = u;
    dfs(v, u);
  }
}

void init() {
  for (int i = 1; (1 << i) <= n; i++) {
    for (int u = 1; u <= n; u++) {
      f[u][i] = f[f[u][i - 1]][i - 1];
    }
  }
}

int getLCA(int u, int v) {
  if (dep[u] < dep[v]) std::swap(u, v);
  for (int i = 20; i >= 0; i--) {
    if (dep[f[u][i]] >= dep[v]) u = f[u][i];
  }
  if (u == v) return u;
  for (int i = 20; i >= 0; i--) {
    if (f[u][i] != f[v][i]) {
      u = f[u][i];
      v = f[v][i];
    }
  }
  return f[u][0];
}

int getDis(int u, int v) {
  if (u == v) return 0;
  int lca = getLCA(u, v);
  return dep[u] + dep[v] - 2 * dep[lca];
}

int getKPar(int u, int k) {
  int cnt = 0;
  while (k) {
    if (k & 1) u = f[u][cnt];
    k >>= 1;
    cnt++;
  } 
  return u;
}

int main() {
  freopen("chase.in", "r", stdin);
  freopen("chase.out", "w", stdout);
  std::ios::sync_with_stdio(false); std::cin.tie(0);
  std::cin >> n >> q;
  for (int i = 1; i <= n - 1; i++) {
    int u, v;
    std::cin >> u >> v;
    g[u].push_back(v);
    g[v].push_back(u);
  }
  dfs(1, 0);
  init();
  while (q--) {
    int s, t, ss;
    std::cin >> s >> t >> ss;
    int disST = getDis(s, t);
    int disSsT = getDis(ss, t);
    if (disSsT > disST) {
      std::cout << disSsT << ' ' << t << '\n';
      continue;
    }
    int disSSs = getDis(s, ss);
    int ans1 = (disSSs + 1) / 2;
    int ans2;
    if (dep[ss] < dep[s]) {
      ans2 = getKPar(s, (disSSs + 1) / 2);
    } else {
      ans2 = getKPar(ss, disSSs / 2);
    }
    std::cout << ans1 << ' ' << ans2 << '\n';
  }
  return 0;
}
```
